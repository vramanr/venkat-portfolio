continuum.ai · test_authoring.py 
from continuum import Agent, LLM
from playwright.sync_api import sync_playwright

agent = Agent(llm=LLM("gpt-4o"), framework="playwright-bdd")

# 1. Read user story  →  generate test cases  →  execute  →  triage
plan = agent.from_story("As a QA lead, I want regression on checkout")
for tc in plan.cases:
    result = agent.run(tc)
    agent.triage(result)  # LLM-summarised defects → Jira
